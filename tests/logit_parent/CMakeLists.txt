cmake_minimum_required(VERSION 3.18)
project(imtg_logit_parent LANGUAGES CXX)

if(NOT IMTG_SOURCE_DIR)
    message(FATAL_ERROR "Set IMTG_SOURCE_DIR")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "Build imtg tests")
set(LOGIT_WITH_CONTEXT ON CACHE BOOL "Parent-selected LogRecord ABI")
option(IMTG_PARENT_PROVIDES_LOGIT "Provide both targets before imtg" ON)
set(IMTG_PARENT_DEPENDENCY_SETTINGS PROVIDED CACHE STRING
    "Provide a target, or request NORMAL/CACHE options before imtg adds it")

if(IMTG_PARENT_DEPENDENCY_SETTINGS STREQUAL "PROVIDED")
    set(LOGIT_CPP_BUILD_TESTS OFF CACHE BOOL "Build provider tests")
    set(LOGIT_CPP_BUILD_EXAMPLES OFF CACHE BOOL "Build provider examples")
    set(TIME_SHIELD_ENABLE_NTP_CLIENT OFF CACHE BOOL "Enable NTP client")
    set(TIME_SHIELD_CPP_BUILD_TESTS OFF CACHE BOOL "Build provider tests")
    set(TIME_SHIELD_CPP_BUILD_EXAMPLES OFF CACHE BOOL "Build provider examples")
    add_subdirectory("${IMTG_SOURCE_DIR}/external/time-shield-cpp" time-shield)
    set(IMTG_TIME_SHIELD_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/unavailable-time-shield" CACHE PATH "Unavailable own provider")
else()
    # Exercise set(CACHE)'s pre-CMP0126 normal-variable removal behavior.
    set(CMAKE_POLICY_DEFAULT_CMP0126 OLD)
    set(parent_settings TIME_SHIELD_ENABLE_NTP_CLIENT
        TIME_SHIELD_CPP_BUILD_TESTS TIME_SHIELD_CPP_BUILD_EXAMPLES
        LOGIT_CPP_BUILD_TESTS LOGIT_CPP_BUILD_EXAMPLES)
    foreach(setting IN LISTS parent_settings)
        if(IMTG_PARENT_DEPENDENCY_SETTINGS STREQUAL "NORMAL")
            set(${setting} ON)
        elseif(IMTG_PARENT_DEPENDENCY_SETTINGS STREQUAL "CACHE")
            set(${setting} ON CACHE BOOL "Parent-selected dependency option")
        else()
            message(FATAL_ERROR "Unknown parent TimeShield settings mode")
        endif()
    endforeach()
endif()
if(IMTG_PARENT_PROVIDES_LOGIT)
    add_subdirectory("${IMTG_SOURCE_DIR}/external/log-it-cpp" log-it-cpp)
    set(IMTG_LOGIT_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/unavailable-logit" CACHE PATH "Unavailable own provider")
endif()
add_subdirectory("${IMTG_SOURCE_DIR}" imtg)

if(NOT LOGIT_WITH_CONTEXT)
    message(FATAL_ERROR "imtg changed the parent logger configuration")
endif()
get_target_property(logit_definitions log-it-cpp::log-it-cpp INTERFACE_COMPILE_DEFINITIONS)
if(NOT "LOGIT_WITH_CONTEXT=1" IN_LIST logit_definitions)
    message(FATAL_ERROR "The shared target lost context support")
endif()
if(NOT IMTG_PARENT_DEPENDENCY_SETTINGS STREQUAL "PROVIDED")
    foreach(setting IN LISTS parent_settings)
        get_directory_property(imtg_setting DIRECTORY "${IMTG_SOURCE_DIR}" DEFINITION ${setting})
        if(NOT ${setting} OR NOT imtg_setting)
            message(FATAL_ERROR "The dependency loader changed the parent's ${setting} setting")
        endif()
    endforeach()
    get_target_property(time_shield_definitions time_shield::time_shield INTERFACE_COMPILE_DEFINITIONS)
    if(NOT "TIME_SHIELD_ENABLE_NTP_CLIENT=$<BOOL:ON>" IN_LIST time_shield_definitions)
        message(FATAL_ERROR "The TimeShield target lost the parent's NTP option: ${time_shield_definitions}")
    endif()
    if(NOT TARGET constants_test OR NOT TARGET date_time_example)
        message(FATAL_ERROR "The TimeShield provider lost the parent's tests/examples options")
    endif()
    if(NOT TARGET memory_logger_backend_test OR NOT TARGET example_logit_basic)
        message(FATAL_ERROR "The LogIt++ provider lost the parent's tests/examples options")
    endif()
endif()

add_executable(imtg_parent_consumer ../logit_install_consumer/main.cpp)
target_link_libraries(imtg_parent_consumer PRIVATE imtg::core log-it-cpp::log-it-cpp)
enable_testing()
add_test(NAME imtg_parent_consumer COMMAND imtg_parent_consumer)
