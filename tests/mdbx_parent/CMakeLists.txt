cmake_minimum_required(VERSION 3.18)
project(imtg_mdbx_parent LANGUAGES C CXX)

set(BUILD_TESTING OFF CACHE BOOL "Build imtg tests")
set(IMTG_BUILD_MDBX ON CACHE BOOL "Build the adapter under test")
set(CMAKE_POLICY_DEFAULT_CMP0126 OLD)
set(parent_settings MDBX_BUILD_SHARED_LIBRARY MDBX_BUILD_TOOLS MDBX_ENABLE_TESTS
    MDBXC_BUILD_EXAMPLES MDBXC_BUILD_TESTS MDBXC_BUILD_BENCHMARKS)
foreach(setting IN LISTS parent_settings)
    if(IMTG_PROVIDER_MODE STREQUAL "NORMAL")
        set(${setting} ON)
    elseif(IMTG_PROVIDER_MODE STREQUAL "CACHE")
        set(${setting} ON CACHE BOOL "Parent-selected dependency option")
    else()
        set(${setting} OFF)
    endif()
endforeach()
if(IMTG_PROVIDER_MODE STREQUAL "CACHE")
    set(MDBXC_DEPS_MODE BUNDLED CACHE STRING "Parent-selected dependency mode")
else()
    if(IMTG_PROVIDER_MODE STREQUAL "NORMAL")
        set(MDBXC_DEPS_MODE SYSTEM CACHE STRING "A distinct parent cache value")
    endif()
    set(MDBXC_DEPS_MODE BUNDLED)
endif()
set(MDBXC_USE_ASAN OFF)
set(IMTG_LOGIT_SOURCE_DIR "${IMTG_LOGIT_PROVIDER_SOURCE_DIR}")
set(IMTG_TIME_SHIELD_SOURCE_DIR "${IMTG_TIME_PROVIDER_SOURCE_DIR}")

if(IMTG_PROVIDER_MODE STREQUAL "SOURCE" OR IMTG_PROVIDER_MODE STREQUAL "NATIVE")
    if(MSVC)
        set(MDBX_C_STANDARD 11)
    endif()
    add_subdirectory("${IMTG_NATIVE_SOURCE_DIR}" native EXCLUDE_FROM_ALL)
    if(MSVC)
        target_compile_options(mdbx-static PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/Zc:__cplusplus>)
    endif()
    if(IMTG_PROVIDER_MODE STREQUAL "SOURCE")
        set(CMAKE_POLICY_DEFAULT_CMP0126 NEW)
        add_subdirectory("${IMTG_CONTAINER_SOURCE_DIR}" containers EXCLUDE_FROM_ALL)
        export(TARGETS mdbx-static mdbx_containers
            NAMESPACE imtg_provider_test::
            FILE "${CMAKE_CURRENT_BINARY_DIR}/package/provider-targets.cmake")
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/package/mdbx_containersConfig.cmake"
            "find_package(Threads REQUIRED)\n"
            "include(\"\${CMAKE_CURRENT_LIST_DIR}/provider-targets.cmake\")\n"
            "add_library(mdbx::mdbx ALIAS imtg_provider_test::mdbx-static)\n"
            "add_library(mdbx_containers::mdbx_containers ALIAS imtg_provider_test::mdbx_containers)\n")
    endif()
elseif(IMTG_PROVIDER_MODE STREQUAL "IMPORTED")
    include("${IMTG_TEST_PACKAGE_DIR}/mdbx_containersConfig.cmake")
elseif(IMTG_PROVIDER_MODE STREQUAL "PACKAGE")
    set(mdbx_containers_DIR "${IMTG_TEST_PACKAGE_DIR}")
    find_package(mdbx_containers CONFIG REQUIRED)
endif()

if(NOT IMTG_PROVIDER_MODE STREQUAL "PINNED")
    set(IMTG_LIBMDBX_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/unavailable-libmdbx")
    set(IMTG_MDBX_CONTAINERS_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/unavailable-containers")
endif()
if(IMTG_PROVIDER_MODE STREQUAL "NATIVE" OR IMTG_PROVIDER_MODE STREQUAL "NORMAL"
        OR IMTG_PROVIDER_MODE STREQUAL "CACHE" OR IMTG_PROVIDER_MODE STREQUAL "MISSING_NATIVE")
    set(IMTG_MDBX_CONTAINERS_SOURCE_DIR "${IMTG_CONTAINER_SOURCE_DIR}")
endif()
if(IMTG_PROVIDER_MODE STREQUAL "NORMAL" OR IMTG_PROVIDER_MODE STREQUAL "CACHE")
    set(IMTG_LIBMDBX_SOURCE_DIR "${IMTG_NATIVE_SOURCE_DIR}")
endif()
# Any implicit lookup would override the explicit source or pinned provider.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/forbidden-package/mdbx_containersConfig.cmake"
    "message(FATAL_ERROR \"Implicit containers package discovery is forbidden\")\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/forbidden-package/MDBXConfig.cmake"
    "message(FATAL_ERROR \"Implicit native package discovery is forbidden\")\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/forbidden-package/mdbxConfig.cmake"
    "message(FATAL_ERROR \"Implicit native package discovery is forbidden\")\n")
set(mdbx_containers_DIR "${CMAKE_CURRENT_BINARY_DIR}/forbidden-package")
set(MDBX_DIR "${CMAKE_CURRENT_BINARY_DIR}/forbidden-package")
set(mdbx_DIR "${CMAKE_CURRENT_BINARY_DIR}/forbidden-package")

if(TARGET mdbx::mdbx)
    set(parent_native_target mdbx::mdbx)
elseif(TARGET mdbx-static)
    set(parent_native_target mdbx-static)
endif()
if(parent_native_target)
    get_target_property(parent_native_options ${parent_native_target} COMPILE_OPTIONS)
    if(NOT parent_native_options)
        set(parent_native_options "")
    endif()
endif()
if(NOT TARGET mdbx_containers::mdbx_containers)
    set(imtg_adds_containers TRUE)
endif()
foreach(setting IN ITEMS ${parent_settings} MDBXC_DEPS_MODE)
    if(DEFINED CACHE{${setting}})
        get_property(parent_cache_${setting} CACHE ${setting} PROPERTY VALUE)
        list(APPEND parent_cached_settings ${setting})
    endif()
endforeach()
add_subdirectory("${IMTG_SOURCE_DIR}" imtg)
foreach(setting IN LISTS parent_cached_settings)
    get_property(actual_cache CACHE ${setting} PROPERTY VALUE)
    if(NOT "${actual_cache}" STREQUAL "${parent_cache_${setting}}")
        message(FATAL_ERROR "imtg changed the parent's cached ${setting}")
    endif()
endforeach()
if(parent_native_target)
    get_target_property(actual_native_options ${parent_native_target} COMPILE_OPTIONS)
    if(NOT actual_native_options)
        set(actual_native_options "")
    endif()
    if(NOT "${actual_native_options}" STREQUAL "${parent_native_options}")
        message(FATAL_ERROR "imtg modified the parent's native target")
    endif()
endif()

if(IMTG_PROVIDER_MODE STREQUAL "NORMAL" OR IMTG_PROVIDER_MODE STREQUAL "CACHE")
    foreach(setting IN LISTS parent_settings)
        get_directory_property(actual DIRECTORY "${IMTG_SOURCE_DIR}/external" DEFINITION ${setting})
        if(NOT ${setting} OR NOT actual)
            message(FATAL_ERROR "imtg changed the parent's ${setting} setting")
        endif()
        if(setting MATCHES "^MDBXC_")
            set(provider_source "${IMTG_CONTAINER_SOURCE_DIR}")
        else()
            set(provider_source "${IMTG_NATIVE_SOURCE_DIR}")
        endif()
        get_directory_property(provider_actual DIRECTORY "${provider_source}" DEFINITION ${setting})
        if(NOT provider_actual)
            message(FATAL_ERROR "The provider did not receive the parent's ${setting} setting")
        endif()
    endforeach()
endif()
get_directory_property(actual_mode DIRECTORY "${IMTG_SOURCE_DIR}/external" DEFINITION MDBXC_DEPS_MODE)
if(NOT actual_mode STREQUAL "BUNDLED" OR NOT MDBXC_DEPS_MODE STREQUAL "BUNDLED")
    message(FATAL_ERROR "imtg changed the parent's dependency mode")
endif()
if(imtg_adds_containers)
    get_directory_property(provider_source DIRECTORY "${IMTG_SOURCE_DIR}/external"
        DEFINITION IMTG_MDBX_CONTAINERS_SOURCE_DIR)
    get_directory_property(provider_mode DIRECTORY "${provider_source}" DEFINITION MDBXC_DEPS_MODE)
    if(NOT provider_mode STREQUAL "BUNDLED")
        message(FATAL_ERROR "The provider did not receive the parent's dependency mode")
    endif()
endif()
get_directory_property(actual_native_path DIRECTORY "${IMTG_SOURCE_DIR}" DEFINITION IMTG_LIBMDBX_SOURCE_DIR)
get_directory_property(actual_containers_path DIRECTORY "${IMTG_SOURCE_DIR}" DEFINITION IMTG_MDBX_CONTAINERS_SOURCE_DIR)
if(NOT "${actual_native_path}" STREQUAL "${IMTG_LIBMDBX_SOURCE_DIR}"
        OR NOT "${actual_containers_path}" STREQUAL "${IMTG_MDBX_CONTAINERS_SOURCE_DIR}")
    message(FATAL_ERROR "imtg changed the parent's source paths")
endif()

add_executable(imtg_mdbx_parent_consumer main.cpp)
target_link_libraries(imtg_mdbx_parent_consumer PRIVATE imtg::mdbx)
enable_testing()
add_test(NAME imtg_mdbx_parent_consumer COMMAND imtg_mdbx_parent_consumer)
set_tests_properties(imtg_mdbx_parent_consumer PROPERTIES
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
